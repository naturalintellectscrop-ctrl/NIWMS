    69→// Trial countdown chip: urgency-colored time remaining with the exact end date
    70→// underneath. Falls back to a plain date for non-trial organizations.
    71→function TrialCountdown({ endsAt, active }: { endsAt: string; active: boolean }) {
    72→  if (!active) {
    73→    return (
    74→      <span className={`inline-flex items-center gap-1.5 ${T.muted}`}>
    75→        <CalendarClock className="h-3.5 w-3.5" />
    76→        {new Date(endsAt).toLocaleDateString()}
    77→      </span>
    78→    )
    79→  }
    80→  const daysLeft = Math.ceil((new Date(endsAt).getTime() - Date.now()) / 86_400_000)
    81→  const expired = daysLeft < 0
    82→  const today = daysLeft === 0
    83→  const soon = daysLeft > 0 && daysLeft <= 3
    84→  const winding = daysLeft > 3 && daysLeft <= 7
    85→  const chip = expired
    86→    ? 'border-[#e2705f]/40 bg-[#e2705f]/10 text-[#e2705f]'
    87→    : today || soon
    88→      ? 'border-[#e2705f]/30 bg-[#e2705f]/5 text-[#f0a08f]'
    89→      : winding
    90→        ? 'border-[#e9b44c]/40 bg-[#e9b44c]/10 text-[#e9b44c]'
    91→        : 'border-white/15 bg-white/5 text-[#d8e2dc]'
    92→  const label = expired ? 'Expired' : today ? 'Ends today' : `${daysLeft}d left`
    93→  return (
    94→    <span className="inline-flex flex-col items-start gap-1">
    95→      <span className={`inline-flex items-center gap-1.5 rounded-full border px-2.5 py-0.5 text-xs font-medium ${chip}`} title={`Trial ends ${new Date(endsAt).toLocaleDateString()}`}>
    96→        <Hourglass className="h-3 w-3" />
    97→        {label}
    98→      </span>
    99→      <span className={`text-[11px] ${T.faint}`}>{new Date(endsAt).toLocaleDateString()}</span>
   100→    </span>
   101→  )
   102→}

   131→function timeAgoLabel(iso: string): string {
   132→  const then = new Date(iso).getTime()
   133→  if (Number.isNaN(then)) return ''
   134→  const seconds = Math.max(0, Math.floor((Date.now() - then) / 1000))
   135→  if (seconds < 60) return 'Just now'
   136→  const minutes = Math.floor(seconds / 60)
   137→  if (minutes < 60) return `${minutes}m ago`
   138→  const hours = Math.floor(minutes / 60)
   139→  if (hours < 24) return `${hours}h ago`
   140→  const days = Math.floor(hours / 24)
   141→  if (days < 7) return `${days}d ago`
   142→  return new Date(iso).toLocaleDateString()
   143→}
