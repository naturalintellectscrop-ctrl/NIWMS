   693→interface QuoteResponse {
   694→  plan: { key: string; name: string; monthlyPrice: number; maxEmployees: number | null; customPricing: boolean }
   695→  interval: BillingInterval

   701→// Interactive quote calculator. The numbers it displays come from the server's
   702→// /api/billing/quote endpoint — the exact totals, VAT, covered period, and
   703→// renewal date a subscriber would actually be charged on that plan + interval.
   704→function PricingCalculator({ interval, plans }: { interval: BillingInterval; plans: PlanCard[] }) {
   705→  const [seats, setSeats] = useState(24)
   706→  const [planKey, setPlanKey] = useState('business')
   707→  const [result, setResult] = useState<QuoteResponse | null>(null)
   708→  const [refreshing, setRefreshing] = useState(false)
   709→
   710→  const pricedPlans = plans.filter((plan) => plan.monthlyPrice !== null)
   711→  // Derived (not synced) selection: if the currently chosen plan cannot hold
   712→  // the requested seats, the calculator automatically falls back to the
   713→  // cheapest plan that can — no cascading setState effects required.
   714→  const selectedPlan =
   715→    pricedPlans.find((plan) => plan.key === planKey && plan.maxEmployees !== null && seats <= plan.maxEmployees) ??
   716→    pricedPlans.find((plan) => plan.maxEmployees !== null && seats <= plan.maxEmployees) ??
   717→    pricedPlans[pricedPlans.length - 1] ??
   718→    pricedPlans[0]
   719→
   720→  useEffect(() => {
   721→    if (!selectedPlan || selectedPlan.monthlyPrice === null) return
   722→    const controller = new AbortController()
   723→    const timer = setTimeout(() => {
   724→      setRefreshing(true)
   725→      fetch(`/api/billing/quote?plan=${selectedPlan.key}&interval=${interval}&seats=${seats}`, { signal: controller.signal })
   726→        .then((response) => (response.ok ? response.json() : null))
   727→        .then((body) => { if (body?.quote) setResult(body) })
   728→        .catch(() => undefined)
   729→        .finally(() => setRefreshing(false))
   730→    }, 200)
   731→    return () => { controller.abort(); clearTimeout(timer) }
   732→  }, [interval, seats, selectedPlan])
   733→
   734→  const quote = result?.quote ?? null
   735→  const trialHref = `/start-free-trial?plan=${selectedPlan?.key ?? 'business'}&interval=${interval}&seats=${seats}`
   736→
   737→  return (
   738→    <div className="reveal reveal-up mt-6 overflow-hidden rounded-2xl border border-[#dce4e1] bg-white shadow-[0_18px_44px_-24px_rgba(18,60,54,0.25)]" data-testid="pricing-calculator">
   739→      <div className="grid gap-0 lg:grid-cols-[1fr_1.1fr]">
   740→        <div className="border-b border-[#dce4e1] p-6 sm:p-8 lg:border-b-0 lg:border-r">
   741→          <p className="flex items-center gap-2 text-xs font-bold uppercase tracking-[0.2em] text-[#c47b32]"><UsersRound className="h-4 w-4" /> Size your team</p>
   742→          <h3 className="mt-3 text-2xl font-bold tracking-[-0.02em] text-[#14201c]">See your real number before you commit.</h3>
   743→          <p className="mt-2 text-sm leading-6 text-[#5b6865]">Slide to your team size. The calculator picks the plan that fits and quotes the exact amount, dates, and renewal.</p>
   744→          <div className="mt-7 flex items-end justify-between"><label htmlFor="seats" className="text-sm font-semibold text-[#304237]">Employees</label><span className="rounded-lg bg-[#e9f0ee] px-3 py-1 text-sm font-bold text-[#123c36]" aria-live="polite">{seats}</span></div>
   745→          <input id="seats" type="range" min={1} max={100} step={1} value={seats} onChange={(event) => setSeats(Number(event.target.value))} className="mt-3 h-2 w-full cursor-pointer accent-[#c47b32]" aria-valuemin={1} aria-valuemax={100} aria-valuenow={seats} />
   746→          <div className="mt-1 flex justify-between text-[11px] text-[#829086]"><span>1</span><span>25</span><span>50</span><span>75</span><span>100+</span></div>
   747→          <div className="mt-5 flex flex-wrap gap-2" role="group" aria-label="Plan">
   748→            {pricedPlans.map((plan) => (
   749→              <button key={plan.key} type="button" aria-pressed={selectedPlan?.key === plan.key} onClick={() => setPlanKey(plan.key)} className={`rounded-full border px-3.5 py-2 text-xs font-semibold transition-colors ${selectedPlan?.key === plan.key ? 'border-[#123c36] bg-[#123c36] text-white' : 'border-[#d2ddda] text-[#5b6865] hover:bg-[#e9f0ee]'}`}>
   750→                {plan.name}
   751→                {plan.maxEmployees !== null && <span className="ml-1 opacity-70">≤{plan.maxEmployees}</span>}
   752→              </button>
   753→            ))}
   754→          </div>
   755→          {seats > 75 && <p className="mt-3 rounded-lg bg-[#fbf0dc] px-3 py-2 text-xs font-medium text-[#8a5a13]" role="note">Above 75 employees our team tailors an Enterprise plan — the quote shows the Professional rate as a reference.</p>}
   756→        </div>
   757→        <div className="bg-[#f7f9f6] p-6 sm:p-8" aria-live="polite">
   758→          <p className="flex items-center justify-between gap-2 text-xs font-bold uppercase tracking-[0.2em] text-[#c47b32]"><span className="flex items-center gap-2"><Receipt className="h-4 w-4" /> Your quote</span><span className={`flex items-center gap-1.5 text-[10px] font-semibold normal-case tracking-normal text-[#829086] ${refreshing ? 'opacity-100' : 'opacity-70'}`}>{refreshing && <span className="inline-block h-1.5 w-1.5 animate-pulse rounded-full bg-[#c47b32]" aria-hidden="true" />}{refreshing ? 'updating…' : 'server-verified'}</span></p>
   759→          {quote ? (
   760→            <>
   761→              <div className="mt-4 flex flex-wrap items-baseline justify-between gap-2">
   762→                <p className="text-sm font-semibold text-[#304237]">{result?.plan.name} · {quote.intervalLabel} billing · {seats} seat{seats === 1 ? '' : 's'}</p>

   773→                  <div className="flex justify-between gap-4 rounded-lg bg-[#e7f0e4] px-3 py-2"><dt className="font-semibold text-[#356247]">You save vs monthly billing</dt><dd className="font-bold text-[#356247]">{ugx(quote.savings)} ({quote.savingsPercent}%)</dd></div>
   774→                )}
   775→              </dl>
   776→              <Link href={trialHref} className="mt-6 flex items-center justify-center rounded-xl bg-[#123c36] px-5 py-3.5 text-sm font-semibold text-white transition-transform hover:-translate-y-0.5" aria-label={`Start free trial on ${result?.plan.name}, billed ${quote.intervalLabel.toLowerCase()}`}>Start with {result?.plan.name} · {quote.intervalLabel} <ArrowRight className="ml-2 h-4 w-4" /></Link>
   777→              <p className="mt-3 text-center text-[11px] text-[#829086]">No payment during the 14-day trial — your billing interval takes effect when you convert.</p>
